# useAverage

Get the average of an array reactively

## Usage

```tsx
import { useAverage } from '@reause/math'

const array = [1, 2, 3]
const averageValue = useAverage(array) // 2
```

```tsx
import { useAverage } from '@reause/math'

const averageValue = useAverage(1, 3) // 2
```
